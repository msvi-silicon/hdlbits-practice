module top_module ( 
    input p1a, p1b, p1c, p1d, p1e, p1f,
    output p1y,
    input p2a, p2b, p2c, p2d,
    output p2y );
    wire N1, N2, N3, N4;
    
    assign N1 = p2a & p2b;
    assign N2 = p2c & p2d;
    assign p2y = N1 | N2;
    
    assign N3 = p1c & p1b & p1a;
    assign N4 = p1f & p1e & p1d;
    assign p1y = N3 | N4;

endmodule
